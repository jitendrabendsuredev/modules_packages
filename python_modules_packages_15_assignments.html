Avoid Circular Import

Student Task

customer.py imports order.py and order.py imports customer.py. Refactor shared name-formatting logic into common.py to avoid circular dependency.

Reference Answer

# common.py
def format_name(name): return name.strip().title()

# customer.py
from common import format_name
def customer_details(name): return "Customer: "+format_name(name)

# order.py
from common import format_name
def order_details(name,oid): return f"{format_name(name)} - Order {oid}"

# main.py
from customer import customer_details
from order import order_details
print(customer_details("rahul"))
print(order_details("rahul",101))
Assignment 15Challenge

Final Project — Online Store

Student Task

Build store package with products.py, billing.py, discount.py and invoice.py. Laptop Bag=2000, Qty=3. Apply 10% discount when subtotal >=5000. Expose functions through __init__.py.

Reference Answer

store/
  __init__.py
  products.py
  billing.py
  discount.py
  invoice.py
main.py

# products.py
def get_product(): return "Laptop Bag",2000
# billing.py
def subtotal(price,qty): return price*qty
# discount.py
def discount(total): return total*0.10 if total>=5000 else 0
# invoice.py
def show(name,total,d):
    print("Product:",name)
    print("Subtotal:",total)
    print("Discount:",d)
    print("Final:",total-d)
# __init__.py
from .products import get_product
from .billing import subtotal
from .discount import discount
from .invoice import show
# main.py
from store import get_product,subtotal,discount,show
name,price=get_product(); total=subtotal(price,3); d=discount(total)
show(name,total,d)
# Final: 5400
